# Goldens whose .out holds facts of the reference build (php 8.5.12, the
# Homebrew build on macOS) that another build of php 8.5.12 prints
# differently: its configuration and the versions and features of the
# libraries it links. `rhp-difftest golden --check --portable` skips them;
# only the reference build may rewrite them. One fixture name per line (path
# below tests/golden, no extension). See docs/testing.md.
#
# Derived by running every golden under Ubuntu's php 8.5 with CI's library
# versions (libxml2 2.15, ICU 78, curl 8.18, OpenSSL 3.5) and with Ubuntu
# 24.04's older ones. The libxml2 entries move to live-php tests once rhp's
# DOM links libxml2 (the golden policy).

core_build_constants    # phpinfo()'s configuration file path
curl_api                # libcurl version: option and info counts, protocol names
curl_setopt_checks      # libcurl version and features (HTTP/3, option ranges)
dom_basic               # libxml2 version: serializer escaping, empty elements
dom_report_build        # libxml2 version: invalid UTF-8 in output (2.15 writes U+FFFD)
dom_save_escape         # libxml2 version: entity/encoding errors, serializer
dom_tree_errors         # libxml2 version: entity reference warnings
dom_xpath               # libxml2 version: XPath error messages
intl_idn                # ICU's UTS #46 data
libxml_errors           # libxml2 version: parser error codes and messages
libxml_parse_edges      # libxml2 version: parser errors, XML declaration checks
openssl_pkey            # OpenSSL's digest list
phpinfo_text            # phpinfo()'s configure command
preg_php_pcre_glue      # PCRE2 version (the r modifier needs 10.43)
simplexml_casts         # libxml2 version: entity reference warnings
xmlreader_basic         # libxml2 version: parser error messages
zip_lifecycle           # libzip version
